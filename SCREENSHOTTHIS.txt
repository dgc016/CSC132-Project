from gpiozero import MotionSensor
from time import sleep

#light to say that there is something in the box
ledpin = 27
sensorpin = 26

Weight.start()
print("start done")

try:
    #detect motion
    while(True):
        hasWeight = Weight.weigh()
        weight = Weight.weighValue()
        if(motionSensor.motion_detected and hasWeight):
            Pushover.run(weight)
            print("Package in mailbox.")

            sleep(60)
           
           
        else:
            GPIO.output(ledpin,GPIO.LOW)
           
        sleep(0.1)
except KeyboardInterrupt:  #CTRL-C
    print("Stopping Program.")
    GPIO.cleanup()

import RPi.GPIO as GPIO
import time
import Pushover

DT = 23
SCK = 20

GPIO.setmode(GPIO.BCM)
GPIO.setup(DT, GPIO.IN)
GPIO.setup(SCK, GPIO.OUT)
GPIO.output(SCK, False)

offset = 0
scale = 1000

def readRaw():
    while GPIO.input(DT) == 1:
        time.sleep(0.0001)

    count = 0
    for _ in range(24):
        GPIO.output(SCK, True)
        count = count << 1
        GPIO.output(SCK, False)
        if GPIO.input(DT):
            count += 1

    # Set gain to 128
    GPIO.output(SCK, True)
    GPIO.output(SCK, False)

    # Convert to signed value
    if count & 0x800000:
        count -= 0x1000000

    return count

def calibrate(samples=20):
    print("Calibrating... Remove all weight.")
    total = 0
    for i in range(samples):
        total += readRaw()
        time.sleep(0.1)
    return total / samples

def start():
    try:
        global offset
        input("Press Enter to calibrate (no weight on scale)...")
        offset = calibrate()
        print("Scale ready.")
    except:
        print("An error has occured.")

def weigh():
    global offset
    raw = readRaw()
    weight = (offset - raw) / scale
    if weight<0:
        weight = 0
    print(f"Weight: {weight:.2f}")
    time.sleep(1)
    if (weight > .5):
        return True
    else:
        return False
    time.sleep(1)

def weighValue():
    global offset
    raw = readRaw()
    weight = (offset - raw) / scale
    return weight
    time.sleep(1)

import http.client, urllib
from PushoverSecrets import token,user
import time

def run(value):
    grams = value*10
    rounded = round(grams,2)
    t = time.localtime()
    currentTime = time.strftime("%H:%M",t)
    conn = http.client.HTTPSConnection("api.pushover.net:443")
    if value > 30:
        conn.request("POST", "/1/messages.json",
            urllib.parse.urlencode({
            "token": token,
            "user": user,
            "title": "SmartMail",
            "message": f"Package detected at {currentTime}, approximately {rounded} grams.",
         }), { "Content-type": "application/x-www-form-urlencoded" })
    else:
        conn.request("POST", "/1/messages.json",
            urllib.parse.urlencode({
            "token": token,
            "user": user,
            "title": "SmartMail",
            "message": f"Light mail detected at {currentTime}, approximately {rounded} grams.",
         }), { "Content-type": "application/x-www-form-urlencoded" })
   
       
    conn.getresponse()
